Code Example

A Codewars kata: sum all natural numbers below a limit that are multiples of 3 or 5.

function solution(number) {
  let sum = 0;

  for (let i = 1; i < number; i++) {
    if (i % 3 === 0 || i % 5 === 0) {
      sum += i;
    }
  }

  return sum;
}

console.log(solution(10)); // 23
console.log(solution(50)); // 543
